  from app.core.database import SessionLocal
  from app.models.user import User
  from app.utils.security import get_password_hash
  from app.services.auth_service import AuthService

  db = SessionLocal()

  try:
      user = db.query(User).filter(User.username == "admin").first()

      if user is None:
          user = AuthService.create_superuser(
              db,
              "admin",
              "admin123",
              "مدیر سیستم",
              "admin@bonyan.ir",
          )
      else:
          user.hashed_password = get_password_hash("admin123")
          user.is_active = True
          user.is_deleted = False
          user.is_superuser = True
          user.deleted_at = None
          db.commit()

      print("username: admin")
      print("password: admin123")
  finally:
      db.close()

ابتدا وضعیت فعلی پوشه bonyancore و تغییرات انجام‌شده تا receipt_allocation.py را بررسی می‌کنم؛ سپس معماری احراز هویت و لاگ سیستم را با الگوهای موجود
  پروژه هماهنگ کرده و تغییرات را مرحله‌به‌مرحله پیاده‌سازی و تست می‌کنم.
وضعیت فعلی نشان می‌دهد بخشی از احراز هویت، کاربر، مدل لاگ و سرویس دریافت قبلاً ایجاد شده اما یکپارچه‌سازی کامل نشده است؛ همچنین فایل‌های زیادی تغییر
  کرده‌اند، بنابراین فقط روی همان تغییرات موجود کار می‌کنم و چیزی را بازنشانی نمی‌کنم. ابتدا وابستگی‌ها، مدل‌ها، main.py، روترها و قالب‌ها را دقیق بررسی
